# 실 구현 예시: MAP-Summary 정합성 자동 검증 import pandas as pd from watchdog.observers import Observer def check_consistency(map_path, sum_path): m = pd.read_excel(map_path) s = pd.read_excel(sum_path) merged = m.merge(s, on=["Lot_ID", "Wafer"], suffixes=("_map", "_sum")) diffs = [] for field in ["PTY", "BIN", "Yield"]: mismatch = merged[merged[f"{field}_map"] != merged[f"{field}_sum"]] diffs.extend(mismatch.to_dict("records")) if diffs: send_alert(diffs, channels=["email", "teams", "kakaowork"]) save_diff_report(diffs, "diff_report.xlsx") return diffs
